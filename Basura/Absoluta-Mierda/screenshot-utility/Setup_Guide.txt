___ (4) Hyprland Config ___

# Place the following in Hyprland Config ( ~/.config/hypr/hyprland.lua )

# OCR scan area and copy text to clipboard (eng+deu+... for more languages)
hl.bind(mainMod .. " + A", hl.dsp.exec_cmd('grim -g "$(slurp)" - | tesseract stdin stdout -l eng | wl-copy'))

# Screenshot area, copy to clipboard, and save to file
hl.bind(mainMod .. " + S", hl.dsp.exec_cmd('grim -g "$(slurp)" - | tee ~/Documents/Screenshots/$(date +%Y-%m-%d_%H-%M-%S).png | wl-copy -t image/png'))

# Pick color from screen and copy hexcode to clipboard
hl.bind(mainMod .. " + D", hl.dsp.exec_cmd("grim -g \"$(slurp -p)\" - | convert - -format '%[pixel:p{0,0}]' info: | tr -d '#' | wl-copy"))

# Screenshot area, open in Satty annotation tool and save edited version to clipboard and file with escape key
hl.bind(mainMod .. " + E", hl.dsp.exec_cmd('grim -g "$(slurp)" - | satty --filename - --output-filename ~/Documents/Screenshots/$(date +%Y-%m-%d_%H-%M-%S).png'))

# Screenshot area with frozen screen. It prevents tabs collapsing on use, but is less smooth. Hyprland does not hide the cursor, but Sway does.
hl.bind(mainMod .. " + W", hl.dsp.exec_cmd("~/.local/bin/screenshot.sh"))

# LaTeX OCR — scan area and copy LaTeX code to clipboard
hl.bind(mainMod .. " + C", hl.dsp.exec_cmd('grim -g "$(slurp)" /tmp/latex-ocr-tmp.png && pix2tex /tmp/latex-ocr-tmp.png | sed "s/^[^:]*: //" | wl-copy && notify-send "LaTeX OCR" "Copied to clipboard"'))

___ (++) Slurp Style Examples ___

# Replace slurp in any keybinding with one of the following to change its appearance.
# The last two hex digits of each color are the alpha/opacity (ff = opaque, 00 = invisible).
# Flags: -b background, -c border, -s selection fill, -w border weight
# The colors can be tested in the terminal

# Example: hl.bind(mainMod .. " + S", hl.dsp.exec_cmd('grim -g "$(slurp -b 1e1e2eaa -c 89b4faff -s 89b4fa22 -w 2)" - | tee ~/Documents/Screenshots/$(date +%Y-%m-%d_%H-%M-%S).png | wl-copy -t image/png'))
